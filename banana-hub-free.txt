--------------------------------------------------------------------------------
-- TẠO NÚT ĐÓNG/MỞ MENU (FLOATING BUTTON)
--------------------------------------------------------------------------------
local ScreenGui = Instance.new("ScreenGui")
local ToggleButton = Instance.new("TextButton")

ScreenGui.Name = "NullRemeumToggle"
ScreenGui.Parent = game:GetService("CoreGui")
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

ToggleButton.Name = "ToggleButton"

ToggleButton.BackgroundColor3 = Color3.fromRGB(255, 255, 255) -- Màu trắng
ToggleButton.Position = UDim2.new(0.1, 0, 0.1, 0)
ToggleButton.Size = UDim2.new(0, 50, 0, 50)
ToggleButton.Font = Enum.Font.GothamBold
ToggleButton.Text = "NULL"

ToggleButton.TextSize = 14.0
ToggleButton.Draggable = true -- Có thể kéo đi khắp màn hình

UICorner.CornerRadius = UDim.new(0, 10)
UICorner.Parent = ToggleButton

ToggleButton.MouseButton1Click:Connect(function()
    game:GetService("VirtualInputManager"):SendKeyEvent(true, Enum.KeyCode.RightControl, false, game)
    game:GetService("VirtualInputManager"):SendKeyEvent(false, Enum.KeyCode.RightControl, false, game)

--------------------------------------------------------------------------------
-- HỆ THỐNG KEY & SCRIPT CHÍNH
--------------------------------------------------------------------------------
local function LoadMainScript()
    local Window = Fluent:CreateWindow({
        Title = "Null Remeum | Premium Edition",
        SubTitle = "v3.0 - All Seas Supported",
        TabWidth = 160,
        Size = UDim2.fromOffset(580, 460),
        Acrylic = false, -- Tắt Acrylic để màu trắng rõ hơn
        Theme = "Light", -- GIAO DIỆN MÀU TRẮNG
        MinimizeKey = Enum.KeyCode.RightControl
    })

    local Tabs = {
        Main = Window:AddTab({ Title = "Main Farm", Icon = "crosshair" }),
        SeaFeatures = Window:AddTab({ Title = "Sea Specials", Icon = "map" }),
        Items = Window:AddTab({ Title = "Items & Boss", Icon = "sword" }),
        Stats = Window:AddTab({ Title = "Stats", Icon = "bar-chart" }),
        Teleport = Window:AddTab({ Title = "Teleport", Icon = "navigation" }),
        Settings = Window:AddTab({ Title = "Settings", Icon = "settings" })
    }

    -- --- CÀI ĐẶT BIẾN ---
    local Options = Fluent.Options

    -- >>> TAB: MAIN FARM <<<
    do
        Tabs.Main:AddParagraph({Title = "Hệ thống Farm", Content = "Tự động hoàn toàn nhiệm vụ và quái."})
        Tabs.Main:AddToggle("AutoFarm", {Title = "Auto Farm Level", Default = false})
        Tabs.Main:AddToggle("FastAttack", {Title = "Fast Attack (Đánh nhanh)", Default = true})
        Tabs.Main:AddToggle("AutoBuso", {Title = "Auto Buso Haki", Default = true})
        
        Tabs.Main:AddDropdown("Weapon", {
            Title = "Chọn Vũ Khí",
            Values = {"Melee", "Sword", "Fruit"},
            Default = 1,
        })
    end

    -- >>> TAB: SEA FEATURES (GOM TẤT CẢ SEA) <<<
    do
        Tabs.SeaFeatures:AddSection("Sea 1")
        Tabs.SeaFeatures:AddButton({Title = "Auto Saber Quest", Callback = function() print("Running Saber...") end})

        Tabs.SeaFeatures:AddSection("Sea 2")
        Tabs.SeaFeatures:AddToggle("AutoRaid", {Title = "Auto Raid (Dungeon)", Default = false})
        Tabs.SeaFeatures:AddToggle("AutoFactory", {Title = "Auto Factory", Default = false})

        Tabs.SeaFeatures:AddSection("Sea 3")
        Tabs.SeaFeatures:AddToggle("AutoBone", {Title = "Auto Farm Bone (Xương)", Default = false})
        Tabs.SeaFeatures:AddToggle("AutoElite", {Title = "Auto Elite Hunter", Default = false})
        Tabs.SeaFeatures:AddToggle("AutoKitsune", {Title = "Auto Kitsune Island", Default = false})
    end

    -- >>> TAB: ITEMS (CDK, SOUL GUITAR...) <<<
    do
        Tabs.Items:AddSection("Legendary Items")
        Tabs.Items:AddToggle("AutoCDK", {Title = "Auto Cursed Dual Katana", Default = false})
        Tabs.Items:AddToggle("AutoSoulGuitar", {Title = "Auto Soul Guitar", Default = false})
        Tabs.Items:AddToggle("AutoTushita", {Title = "Auto Tushita Quest", Default = false})
        Tabs.Items:AddToggle("AutoYama", {Title = "Auto Yama Quest", Default = false})
    end

    -- >>> TAB: TELEPORT <<<
    do
        Tabs.Teleport:AddDropdown("TeleportIsland", {
            Title = "Dịch chuyển Đảo",
            Values = {"Sea 1", "Sea 2", "Sea 3", "Mansion", "Cafe", "Hydra Island"},
            Callback = function(v) print("Teleport to " .. v) end
        })
        Tabs.Teleport:AddToggle("ESPFruit", {Title = "ESP Trái Ác Quỷ", Default = false})
    end

    -- >>> TAB: SETTINGS <<<
    do
        Tabs.Settings:AddSection("Cấu hình")
        Tabs.Settings:AddButton({
            Title = "Hủy Key / Đăng xuất",
            Callback = function() Window:Destroy() end
        })
    end

    Window:SelectTab(1)
    Fluent:Notify({Title = "Null Remeum", Content = "Đã nạp thành công giao diện White Edition!", Duration = 5})
end

--------------------------------------------------------------------------------
-- CỬA SỔ KEY SYSTEM (XUẤT HIỆN ĐẦU TIÊN)
--------------------------------------------------------------------------------
local KeyWindow = Fluent:CreateWindow({
    Title = "Null Remeum | Key System",
    SubTitle = "Vui lòng nhập Key",
    TabWidth = 160,
    Size = UDim2.fromOffset(450, 300),
    Acrylic = false,
    Theme = "Light", -- Màu trắng
    MinimizeKey = Enum.KeyCode.RightControl
})

local KeyTab = KeyWindow:AddTab({ Title = "Hệ thống Key", Icon = "key" })

local InputKey = ""
local MyKey = "NULL-PREMIUM" -- Key mặc định

KeyTab:AddInput("KeyInput", {
    Title = "Nhập Key của bạn",
    Default = "",
    Placeholder = "Dán key tại đây...",
    Callback = function(Value) InputKey = Value end
})

KeyTab:AddButton({
    Title = "Kiểm tra Key",
    Callback = function()
        if InputKey == MyKey then
            Fluent:Notify({Title = "Thành công", Content = "Đang mở Menu chính...", Duration = 3})
            KeyWindow:Destroy()
            LoadMainScript()
        else
            Fluent:Notify({Title = "Sai Key", Content = "Vui lòng kiểm tra lại key của bạn!", Duration = 3})
        end
    end
})

KeyTab:AddButton({
    Title = "Lấy Key (Get Key)",
    Callback = function()
        setclipboard("https://link-lay-key.com/null-remeum")
        Fluent:Notify({Title = "Thông báo", Content = "Đã copy link lấy key vào bộ nhớ!", Duration = 5})
    end
})

-- Thông báo ban đầu
Fluent:Notify({
    Title = "Null Remeum",
    Content = "Bấm vào nút 'NULL' màu trắng để đóng/mở Menu!",
    Duration = 7
})
