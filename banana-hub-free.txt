    VirtualUser:Button2Down(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
    task.wait(1)
    VirtualUser:Button2Up(Vector2.new(0,0), workspace.CurrentCamera.CFrame)
end)

--------------------------------------------------------------------------------
-- LOGIC VẬN HÀNH (CORE LOGIC)
--------------------------------------------------------------------------------

-- 1. Đánh Quái & Fast Attack
task.spawn(function()
    while task.wait() do
        if Options.AutoFarm.Value or Options.AutoFarmBone.Value or Options.AutoFarmNearest.Value then
            pcall(function()
                -- Tự trang bị vũ khí
                local tool = LP.Backpack:FindFirstChild(_G.SelectWeapon) or LP.Character:FindFirstChild(_G.SelectWeapon)
                if tool then LP.Character.Humanoid:EquipTool(tool) end
                
                -- Fast Attack
                VirtualUser:CaptureController()
                VirtualUser:ClickButton1(Vector2.new(0,0))
            end)
        end
    end
end)

-- 2. Tự Động Gacha & Store
task.spawn(function()
    while task.wait(5) do
        if Options.AutoGacha.Value then
            CommF:InvokeServer("Cousin", "Buy")
        end
        if Options.AutoStore.Value then
            for _, v in pairs(LP.Backpack:GetChildren()) do
                if v:IsA("Tool") and v.ToolTip == "Blox Fruit" then
                    CommF:InvokeServer("StoreFruit", v:GetAttribute("OriginalName") or v.Name, v)
                end
            end
        end
    end
end)

-- 3. Auto Stats
task.spawn(function()
    while task.wait(0.5) do
        if Options.AutoFarm.Value then
            local points = LP.Data.Points.Value
            if points > 0 then
                -- Ưu tiên cộng đều Melee & Defense
                CommF:InvokeServer("AddPoint", "Melee", 1)
                CommF:InvokeServer("AddPoint", "Defense", 1)
            end
        end
    end
end)
