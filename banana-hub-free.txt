-- 1. LOAD THƯ VIỆN FLUENT UI
local Fluent = loadstring(game:HttpGet("https://github.com/dawid-scripts/Fluent/releases/latest/download/main.lua"))()
local SaveManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/SaveManager.lua"))()
local InterfaceManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/InterfaceManager.lua"))()

-- 2. TẠO NÚT ĐÓNG/MỞ MENU TRÊN MÀN HÌNH (CHO MOBILE)
do
    if game.CoreGui:FindFirstChild("NullRemeumToggle") then
        game.CoreGui.NullRemeumToggle:Destroy()
    end

    local ScreenGui = Instance.new("ScreenGui")
    local ToggleButton = Instance.new("TextButton")
    local UICorner = Instance.new("UICorner")
    local UIStroke = Instance.new("UIStroke")

    ScreenGui.Name = "NullRemeumToggle"
    ScreenGui.Parent = game:GetService("CoreGui")
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

    ToggleButton.Name = "ToggleButton"
    ToggleButton.Parent = ScreenGui
    ToggleButton.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    ToggleButton.Position = UDim2.new(0.1, 0, 0.1, 0)
    ToggleButton.Size = UDim2.new(0, 45, 0, 45)
    ToggleButton.Font = Enum.Font.GothamBold
    ToggleButton.Text = "N"
    ToggleButton.TextColor3 = Color3.fromRGB(30, 30, 30)
    ToggleButton.TextSize = 20.0
    ToggleButton.Draggable = true
    ToggleButton.AutoButtonColor = true

    UICorner.CornerRadius = UDim.new(0, 10)
    UICorner.Parent = ToggleButton

    UIStroke.Parent = ToggleButton
    UIStroke.Thickness = 2
    UIStroke.Color = Color3.fromRGB(200, 200, 200)

    -- Logic: Giả lập phím RightControl để đóng mở UI Fluent
    ToggleButton.MouseButton1Click:Connect(function()
        game:GetService("VirtualInputManager"):SendKeyEvent(true, Enum.KeyCode.RightControl, false, game)
        game:GetService("VirtualInputManager"):SendKeyEvent(false, Enum.KeyCode.RightControl, false, game)
    end)
end

-- HỆ THỐNG KEY SYSTEM (CHẠY ĐẦU TIÊN)
--------------------------------------------------------------------------------
local function LoadKeySystem()
    local KeyWindow = Fluent:CreateWindow({
        Title = "Null Remeum | Key System",
        SubTitle = "Security Check",
        TabWidth = 160,
        Size = UDim2.fromOffset(500, 320),
        Acrylic = false,
        Theme = "Light",
        MinimizeKey = Enum.KeyCode.RightControl
    })

    local KeyTab = KeyWindow:AddTab({ Title = "Verify Key", Icon = "lock" })
    local KeyInputVal = ""
    local TrueKey = "NULL-PREMIUM" -- ĐẶT KEY CỦA BẠN TẠI ĐÂY

    KeyTab:AddParagraph({
        Title = "Hướng dẫn",
        Content = "Vui lòng lấy Key để sử dụng Script. Key: " .. TrueKey
    })

    KeyTab:AddInput("InputKey", {
        Title = "Nhập Key",
        Default = "",
        Placeholder = "Paste Key Here...",
        Callback = function(Value)
            KeyInputVal = Value
        end
    })

    KeyTab:AddButton({
        Title = "Kiểm tra Key (Check Key)",
        Callback = function()
            if KeyInputVal == TrueKey then
                Fluent:Notify({Title = "Thành công", Content = "Key chính xác! Đang tải Menu...", Duration = 3})
                KeyWindow:Destroy() -- Tắt bảng Key
                wait(0.5)
                LoadMainScript() -- Mở bảng Farm chính
            else
                Fluent:Notify({Title = "Lỗi", Content = "Key sai hoặc chưa nhập!", Duration = 3})
            end
        end
    })

    KeyTab:AddButton({
        Title = "Lấy Key (Get Key Link)",
        Callback = function()
            setclipboard("https://example.com/getkey")
            Fluent:Notify({Title = "Link Copied", Content = "Đã copy link lấy key!", Duration = 3})
        end
    })
end

-- BẮT ĐẦU CHẠY SCRIPT
LoadKeySystem()
